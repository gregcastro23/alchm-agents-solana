      ;(function () {

    vec2 starUV = uv * 100.0;

    vec2 fpos = fract(starUV);
    float h = hash(ipos);
    if (h > 0.98) {
        float twinkle = sin(u_time * 2.0 + h * 6.28) * 0.5 + 0.5;
        float size = 0.1 * h;
        float star = smoothstep(size, 0.0, length(fpos - 0.5)) * twinkle;
        color += star * vec3(0.9, 0.9, 1.0);
    }
    gl_FragColor = vec4(color, 1.0);
}`
