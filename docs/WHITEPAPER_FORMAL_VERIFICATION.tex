\begin{lemma}[Universal AMM Ratio Lemma]
\label{lem:amm_ratio}
For any valid swap input $\Delta x$, reserves $R_{\text{in}}, R_{\text{out}} > 0$, and fee parameter $\text{feeBps} \le \text{BPS}$:
\begin{equation}
    \text{outAmt} \cdot R_{\text{in}} \le \Delta x \cdot R_{\text{out}}
\end{equation}
\end{lemma}
\begin{proof}
By integer division floor on the deployed formula:
$$\text{outAmt} \cdot (R_{\text{in}} + \text{inWithFee}) \le \text{inWithFee} \cdot R_{\text{out}}$$
Since $R_{\text{in}} \le R_{\text{in}} + \text{inWithFee}$:
$$\text{outAmt} \cdot R_{\text{in}} \le \text{inWithFee} \cdot R_{\text{out}}$$
Since $\text{inWithFee} = \lfloor \Delta x \cdot (\text{BPS} - \text{feeBps}) / \text{BPS} \rfloor \le \Delta x$:
$$\text{outAmt} \cdot R_{\text{in}} \le \Delta x \cdot R_{\text{out}}$$
Machine-checked in \texttt{Proofs/ConstellationAMM.lean} (\texttt{getAmountOut\_mul\_reserveIn\_le}).
\end{proof}

\begin{theorem}[Cyclic Multi-Hop Swap Conservation (No-Infinite-Mint)]
\label{thm:amm_cycle}
Consider a 3-hop cyclic swap sequence $A \xrightarrow{\Delta x_1} B \xrightarrow{\Delta x_2} C \xrightarrow{\Delta x_3} A$ through distinct elemental pools $(R_A^{AB}, R_B^{AB})$, $(R_B^{BC}, R_C^{BC})$, and $(R_C^{CA}, R_A^{CA})$. Under the no-arbitrage pool reserve condition:
\begin{equation}
    R_B^{AB} \cdot R_C^{BC} \cdot R_A^{CA} \le R_A^{AB} \cdot R_B^{BC} \cdot R_C^{CA}
\end{equation}
the final output amount $\text{out}_{A,\text{final}}$ satisfies:
\begin{equation}
    \text{out}_{A,\text{final}} \le \text{in}_{A,\text{initial}}
\end{equation}
\end{theorem}
\begin{proof}
Applying Lemma~\ref{lem:amm_ratio} sequentially across all 3 hops and multiplying the inequalities yields:
$$\text{out}_{A,\text{final}} \cdot R_A^{CA} R_C^{BC} R_B^{AB} \le \text{in}_{A,\text{initial}} \cdot R_C^{CA} R_B^{BC} R_A^{AB}$$
Under the hypothesis, the product of input reserves is greater than or equal to the output reserves, establishing $\text{out}_{A,\text{final}} \le \text{in}_{A,\text{initial}}$. Machine-checked in \texttt{Proofs/ConstellationAMM.lean} (Theorem 5, \texttt{no\_infinite\_mint\_cycle}).
\end{proof}
